\begin{frame}{从具体任务选题，不从技术名词选题}
  \begin{center}
  \begin{tikzpicture}[node distance=0.38cm and 0.40cm]
    \node[coursebox,minimum width=2.25cm] (who) {谁在使用？};
    \node[coursebox,minimum width=2.25cm,right=of who] (scene) {什么场景？};
    \node[actionbox,minimum width=2.25cm,right=of scene] (task) {完成什么任务？};
    \node[safebox,minimum width=2.25cm,right=of task] (proof) {怎样证明？};
    \node[coursebox,minimum width=2.25cm,right=of proof] (scope) {两周能否做完？};
    \draw[flowarrow](who)--(scene);
    \draw[flowarrow](scene)--(task);
    \draw[flowarrow](task)--(proof);
    \draw[flowarrow](proof)--(scope);
  \end{tikzpicture}
  \end{center}
  \vspace{0.55em}
  \begin{block}{推荐的一句话表达}
    面向【某类用户】，在【具体场景】下，帮助其完成【可观察任务】，以【数据、工具结果或操作记录】证明任务完成。
  \end{block}
  \begin{alertblock}{不建议}
    “做一个万能座舱助手”范围过大；“调用一次大模型回答问题”又缺少状态、行动和结果验证。
  \end{alertblock}
\end{frame}

\begin{frame}{用五个问题判断这个题是否适合课程项目}
  \small
  \begin{tabularx}{\textwidth}{>{\bfseries}p{1.0cm} p{5.5cm} X}
    \toprule
    序号 & 判断问题 & 合格表现 \\
    \midrule
    1 & 用户和使用场景是否明确？ & 能描述一次真实、具体的使用过程 \\
    2 & 是否需要多步处理或持续状态？ & 不只是一次问答，存在记忆、工具或操作 \\
    3 & 结果是否能够验证？ & 有数据、日志、界面状态或明确规则 \\
    4 & 失败时能否安全停止或降级？ & 能拒绝、澄清、确认或只做模拟 \\
    5 & 是否能在两周内完成最小版本？ & 先跑通一条主流程，再做一个扩展点 \\
    \bottomrule
  \end{tabularx}
  \vspace{0.45em}
  \begin{center}
    \tagpill{五问都能回答，再进入方案设计}
  \end{center}
\end{frame}

\begin{frame}{选题可以从三项实践项目中自然延伸}
  \small
  \begin{tabularx}{\textwidth}{>{\bfseries}p{3.0cm} p{4.2cm} X}
    \toprule
    建议方向 & 最小任务 & 可增加的一个亮点 \\
    \midrule
    个性化座舱助理 & 记住并更新用户偏好 & 用户查看、修改和删除记忆 \\
    车辆健康问答 & 调用车辆工具回答状态问题 & 组合多个工具或解释故障证据 \\
    多模态界面助手 & 根据界面信息生成安全操作建议 & 低置信度澄清或高风险确认 \\
    \bottomrule
  \end{tabularx}
  \vspace{0.55em}
  \begin{block}{选择建议}
    优先复用一个已经跑通的实践项目，再增加一个清楚、可测试的改进点；不要同时重做记忆、工具、多模态和前端。
  \end{block}

\end{frame}

\begin{frame}{最终目标：跑通一条可验证、可解释的任务链}
  \begin{center}
  \begin{tikzpicture}[node distance=0.40cm and 0.36cm]
    \node[coursebox,minimum width=2.35cm] (input) {明确输入};
    \node[actionbox,minimum width=2.35cm,right=of input] (process) {智能体处理};
    \node[actionbox,minimum width=2.35cm,right=of process] (action) {回答或模拟行动};
    \node[safebox,minimum width=2.35cm,right=of action] (check) {检查结果};
    \node[coursebox,minimum width=2.35cm,right=of check] (record) {记录证据};
    \draw[flowarrow](input)--(process);
    \draw[flowarrow](process)--(action);
    \draw[flowarrow](action)--(check);
    \draw[flowarrow](check)--(record);
  \end{tikzpicture}
  \end{center}
  \vspace{0.55em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{至少演示}
        \begin{itemize}
          \item 一条正常完成路径；
          \item 一条输入错误或工具失败路径；
          \item 一条高风险拒绝、确认或降级路径。
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{至少留下}
        \begin{itemize}
          \item 最终输出或界面状态；
          \item 工具结果、日志或数据记录；
          \item 测试输入、预期与实际结果。
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{实现时只做“一个基础主线 + 一个明确改进”}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{基础主线可以复用}
        \begin{itemize}
          \item 实践项目一：记忆保存与更新；
          \item 实践项目二：车辆工具调用；
          \item 实践项目三：界面理解与安全操作建议。
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{改进点只选一个}
        \begin{itemize}
          \item 更可靠的事实冲突处理；
          \item 多工具组合与失败恢复；
          \item 更明确的风险确认；
          \item 更丰富但仍可验证的界面任务。
        \end{itemize}
      \end{block}
  \end{columns}
  \begin{alertblock}{范围控制}
    先保证主流程可以重复运行，再增加亮点；不能验证的功能不计入最终目标。
  \end{alertblock}
\end{frame}

\begin{frame}{项目设计必须注意四件事}
  \small
  \begin{tabularx}{\textwidth}{>{\bfseries}p{2.3cm} X X}
    \toprule
    注意点 & 需要提前写清 & 检查方式 \\
    \midrule
    范围 & 做什么、不做什么 & 两周计划是否可执行 \\
    证据 & 回答或操作依据来自哪里 & 结果中能否找到数据或日志 \\
    安全 & 哪些情况要拒绝、确认或降级 & 加入错误、攻击和高风险用例 \\
    复现 & 环境、数据、命令和版本 & 换一台机器能否按 README 跑通 \\
    \bottomrule
  \end{tabularx}
  \begin{alertblock}{座舱项目边界}
    课程项目默认使用模拟数据和模拟执行，不连接真实车辆控制接口；涉及个人或车辆数据时必须说明来源并完成脱敏。
  \end{alertblock}
\end{frame}
