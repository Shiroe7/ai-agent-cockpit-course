\begin{frame}{本节课围绕一个问题：怎样让智能体真正“记得住”？}
  \begin{columns}[T,onlytextwidth]
    \column{0.57\textwidth}
      \begin{block}{学习目标}
        \begin{enumerate}
          \item 说清普通聊天与有状态智能体的差别；
          \item 认识智能体框架中的五个核心模块；
          \item 理解记忆的写入、检索、更新与隔离；
          \item 完成一个可跨会话记忆用户偏好的智能体。
        \end{enumerate}
      \end{block}
    \column{0.39\textwidth}
      \begin{alertblock}{贯穿本课的场景}
        用户第一次说“我喜欢舒适模式”，关闭程序后再次询问：
        \begin{center}
          \textbf{“我平时喜欢什么驾驶模式？”}
        \end{center}
        系统能否给出有依据的回答？
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{普通聊天只看当前输入，项目目标是跨会话保持事实}
  \begin{center}
  \begin{tikzpicture}[node distance=0.45cm and 0.55cm]
    \node[coursebox,minimum width=3.0cm] (say) {会话 A\\“我喜欢舒适模式”};
    \node[actionbox,minimum width=3.0cm,right=of say] (write) {识别并保存\\偏好事实};
    \node[safebox,minimum width=3.0cm,right=of write] (close) {程序关闭\\再次启动};
    \node[coursebox,minimum width=3.0cm,right=of close] (recall) {会话 B\\检索后回答};
    \draw[flowarrow](say)--(write);
    \draw[flowarrow](write)--(close);
    \draw[flowarrow](close)--(recall);
  \end{tikzpicture}
  \end{center}
  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{普通聊天容易出现的问题}
        \begin{itemize}
          \item 新会话看不到旧消息；
          \item 全量回放越来越长；
          \item 新旧偏好可能相互冲突。
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{本项目希望做到}
        \begin{itemize}
          \item 只保存值得长期保留的事实；
          \item 回答前按用户和会话检索；
          \item 偏好变化时更新旧事实。
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{智能体框架把一次回答组织成可重复的运行闭环}
  \begin{center}
  \begin{tikzpicture}[node distance=0.38cm and 0.42cm]
    \node[coursebox,minimum width=2.25cm] (input) {1 接收任务};
    \node[coursebox,minimum width=2.25cm,right=of input] (read) {2 读取状态};
    \node[actionbox,minimum width=2.25cm,right=of read] (decide) {3 模型决策};
    \node[actionbox,minimum width=2.25cm,right=of decide] (act) {4 回答/行动};
    \node[safebox,minimum width=2.25cm,right=of act] (record) {5 验证并记录};
    \draw[flowarrow](input)--(read);
    \draw[flowarrow](read)--(decide);
    \draw[flowarrow](decide)--(act);
    \draw[flowarrow](act)--(record);
    \draw[flowarrow](record.south)--++(0,-0.5)-|(read.south);
  \end{tikzpicture}
  \end{center}
  \vspace{0.55em}
  \begin{block}{为什么它不只是 Chatbot}
    下一轮处理会使用上一轮留下的状态和证据；失败时也能拒绝、重试或停止，而不是只生成一段“看似合理”的文本。
  \end{block}
  \begin{alertblock}{本课聚焦}
    五个模块共同形成智能体；实践项目一先把其中的“状态与记忆”做扎实。
  \end{alertblock}
\end{frame}

\begin{frame}{记忆不是聊天记录堆积，而是受控的事实更新}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{每轮对话只做四个判断}
        \begin{enumerate}
          \item 这句话是否包含可保存事实？
          \item 应当新增、更新、删除还是保持？
          \item 事实属于哪个用户和会话？
          \item 回答当前问题需要检索哪些事实？
        \end{enumerate}
      \end{block}
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=0.38cm]
        \node[coursebox,minimum width=4.2cm] (utterance) {用户话语};
        \node[actionbox,minimum width=4.2cm,below=of utterance] (extract) {抽取候选事实};
        \node[safebox,minimum width=4.2cm,below=of extract] (check) {校验来源与冲突};
        \node[coursebox,minimum width=4.2cm,below=of check] (store) {写入/更新 SQLite};
        \draw[flowarrow](utterance)--(extract);
        \draw[flowarrow](extract)--(check);
        \draw[flowarrow](check)--(store);
      \end{tikzpicture}
      \end{center}
  \end{columns}
  \vspace{-0.55em}
  \begin{alertblock}{课堂判断}
    “我以前喜欢运动模式，但现在更喜欢舒适模式”应更新原偏好，而不是并列保存两个互相冲突的事实。
  \end{alertblock}
\end{frame}

\begin{frame}{实践项目一：目标、思路与成功标准}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}{项目目标}
        构建一个命令行对话智能体，使它能在不同会话之间记住、检索并更新用户偏好，同时保证不同会话数据不串联。
      \end{block}
      \begin{block}{实现思路}
        \begin{enumerate}
          \item 保存每轮消息；
          \item 从陈述中抽取稳定事实；
          \item 回答前检索相关事实；
          \item 新事实与旧事实冲突时更新。
        \end{enumerate}
      \end{block}
    \column{0.47\textwidth}
      \begin{block}{成功标准}
        \begin{itemize}
          \item 同一会话能承接上下文；
          \item 新会话能召回用户偏好；
          \item 偏好变化后只保留新值；
          \item 不同会话互不串数据；
          \item 事实来源可查看、结果可测试。
        \end{itemize}
      \end{block}
      \tagpill[CourseGreen]{默认无需外部 API}
  \end{columns}

\end{frame}

\begin{frame}{长期记忆的主要风险，也对应实验中的检查点}
  \small
  \begin{tabularx}{\textwidth}{>{\bfseries}p{2.5cm} X X}
    \toprule
    风险 & 可能后果 & 本项目的控制方法 \\
    \midrule
    错误写入 & 模型猜测长期影响后续回答 & 只保存可识别字段，保留来源 \\
    事实冲突 & 同一偏好出现多个版本 & 新事实更新旧事实，不盲目追加 \\
    会话串扰 & 把其他会话内容当成当前用户事实 & 使用用户与会话标识隔离 \\
    隐私泄露 & 敏感信息被长期保存或输出 & 敏感字段默认不持久化 \\
    删除失效 & 用户要求删除后仍可召回 & 提供查看、修订和删除路径 \\
    \bottomrule
  \end{tabularx}
  \begin{alertblock}{原则}
    模型可以建议“写入什么”，应用必须决定“是否允许写入”。
  \end{alertblock}

\end{frame}
