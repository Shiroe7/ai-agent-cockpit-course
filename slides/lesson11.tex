\begin{frame}{实践项目二：目标、思路与成功标准}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}{项目目标}
        让智能体根据用户问题选择正确的车辆数据工具，生成合法参数，并依据工具返回结果作答。
      \end{block}
      \begin{block}{实现思路}
        文本问题 $\rightarrow$ 选择工具 $\rightarrow$ 校验参数 $\rightarrow$ 查询本地 JSON $\rightarrow$ 返回数据与证据。
      \end{block}
    \column{0.47\textwidth}
      \begin{block}{成功标准}
        \begin{itemize}
          \item 正确选择车辆状态、胎压、告警或故障码工具；
          \item 非法参数被明确拒绝；
          \item 成功和失败结果都结构化；
          \item 没有工具证据时不编造车辆数据。
        \end{itemize}
      \end{block}
      \tagpill[CourseGreen]{本地模拟数据，可重复测试}
  \end{columns}

\end{frame}

  \vspace{0.15em}
  \small
  \begin{tabularx}{\textwidth}{>{\bfseries}p{2.5cm} X}
    allowlist & 只允许 \texttt{tap/set\_value/refuse} 等已定义动作；\\
    target existence & 目标必须存在于当前界面元素列表；\\
    confidence gate & 低置信度不执行，改为澄清或拒绝；\\
    confirmation & 涉及支付、账户、驾驶关键设置时要求确认；\\
    dry-run & 教学代码只模拟执行并记录审计日志。\\

\begin{frame}{多模态安全：图像内容也可能是“提示注入”}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{攻击与失效}
        \begin{itemize}
          \item 图中隐藏“忽略规则”的文本；
          \item 看错目标、坐标或界面状态；
          \item 置信度虚高；
          \item 连续点击造成不可逆后果；
          \item 视觉建议越权变成控制命令。
        \end{itemize}
      \end{block}
    \column{0.49\textwidth}
      \begin{block}{防护与降级}
        \begin{itemize}
          \item 图像文本只视为不可信数据；
          \item 目标必须来自可信 UI 树/清单；
          \item 关键动作二次确认；
          \item 每步重新观察，不盲目连击；
          \item 不确定时停止并向用户澄清。
        \end{itemize}
      \end{block}
  \end{columns}
  \begin{alertblock}{座舱边界}
    VLM 可用于感知与建议；真实车辆控制必须经过独立、确定性的安全控制链。
  \end{alertblock}
  \projectref{实践项目三安全设计：《大模型安全与合规》p.13、42、51、54-55；《自动驾驶和Alpamayo》p.29、44-46}
\end{frame}
